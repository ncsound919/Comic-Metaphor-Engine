```python
#!/usr/bin/env python3
"""
Update knowledge_base.json with new protocols from expanded text files.

Reads:
- avengers_cosmic_protocols_complete.txt
- character_deep_dives_complete.txt

Parses protocols and updates:
- processed/knowledge_base.json

Maintains existing 76 protocols while adding 27 new ones.
"""

import json
import re
from pathlib import Path
from typing import Dict, List, Any


def parse_protocol_file(filepath: str) -> List[Dict[str, Any]]:
    """Parse protocol text file into structured data"""
    protocols = []
    
    with open(filepath, 'r', encoding='utf-8') as f:
        content = f.read()
    
    # Split by protocol marker (##)
    protocol_blocks = re.split(r'\n## ', content)
    
    for block in protocol_blocks[1:]:  # Skip first empty split
        protocol = {}
        
        # Extract title
        title_match = re.search(r'^(.+?)(?:\n|$)', block)
        if title_match:
            protocol['name'] = title_match.group(1).strip()
        
        # Extract sections
        protocol['storyline'] = extract_section(block, 'Storyline Summary|Character Background')
        protocol['core_metaphor'] = extract_section(block, 'Core Business Metaphor|Core Business Pattern')
        
        # Extract dimensions
        protocol['d1_bio'] = extract_section(block, 'D1.*Bio/Internal')
        protocol['d2_tech'] = extract_section(block, 'D2.*Tech/External')
        protocol['d3_eco'] = extract_section(block, 'D3.*Eco/Resources')
        protocol['d4_cosmic'] = extract_section(block, 'D4.*Cosmic/Limit')
        
        # Extract lists
        protocol['business_applications'] = extract_list(block, 'Business Applications|Business Parallels')
        protocol['warning_signs'] = extract_list(block, 'Warning Signs|When You See This Pattern')
        protocol['counter_strategies'] = extract_list(block, 'Counter-Strategies|How to Address')
        
        # Extract metadata
        protocol['quality_score'] = extract_float(block, 'Quality Score')
        protocol['risk_severity'] = extract_value(block, 'Risk Severity')
        protocol['time_horizon'] = extract_value(block, 'Time Horizon')
        protocol['industry_tags'] = extract_tags(block, 'Industry Tags')
        
        protocols.append(protocol)
    
    return protocols


def extract_section(text: str, section_name: str) -> str:
    """Extract text from a section"""
    pattern = rf'\*\*{section_name}\*\*:\s*\n(.+?)(?=\n\*\*|\n##|$)'
    match = re.search(pattern, text, re.DOTALL | re.IGNORECASE)
    return match.group(1).strip() if match else ""


def extract_list(text: str, section_name: str) -> List[str]:
    """Extract bulleted or numbered list from section"""
    section = extract_section(text, section_name)
    items = []
    
    for line in section.split('\n'):
        line = line.strip()
        # Match bullet points or numbers
        match = re.match(r'^[\d\-\*\•]\s*\.?\s*(.+)', line)
        if match:
            items.append(match.group(1).strip())
    
    return items


def extract_float(text: str, field_name: str) -> float:
    """Extract float value from field"""
    pattern = rf'\*\*{field_name}\*\*:\s*([\d\.]+)'
    match = re.search(pattern, text, re.IGNORECASE)
    return float(match.group(1)) if match else 0.0


def extract_value(text: str, field_name: str) -> str:
    """Extract single value from field"""
    pattern = rf'\*\*{field_name}\*\*:\s*([^\n]+)'
    match = re.search(pattern, text, re.IGNORECASE)
    return match.group(1).strip() if match else ""


def extract_tags(text: str, field_name: str) -> List[str]:
    """Extract comma-separated tags"""
    value = extract_value(text, field_name)
    return [tag.strip() for tag in value.split(',') if tag.strip()]


def update_knowledge_base(avengers_file: str, characters_file: str, kb_file: str):
    """Update knowledge base with new protocols"""
    
    # Load existing knowledge base
    kb_path = Path(kb_file)
    if kb_path.exists():
        with open(kb_path, 'r', encoding='utf-8') as f:
            knowledge_base = json.load(f)
    else:
        knowledge_base = {
            "version": "2.0",
            "updated": "",
            "protocols": []
        }
    
    # Parse new protocols
    print(f"Parsing {avengers_file}...")
    avengers_protocols = parse_protocol_file(avengers_file)
    print(f"  Found {len(avengers_protocols)} Avengers protocols")
    
    print(f"Parsing {characters_file}...")
    character_protocols = parse_protocol_file(characters_file)
    print(f"  Found {len(character_protocols)} Character protocols")
    
    # Add category tags
    for p in avengers_protocols:
        p['category'] = 'Avengers Cosmic'
    
    for p in character_protocols:
        p['category'] = 'Character Deep Dive'
    
    # Merge with existing (avoid duplicates by name)
    existing_names = {p.get('name', '') for p in knowledge_base['protocols']}
    
    new_count = 0
    for protocol in avengers_protocols + character_protocols:
        if protocol['name'] not in existing_names:
            knowledge_base['protocols'].append(protocol)
            new_count += 1
    
    # Update metadata
    from datetime import datetime
    knowledge_base['updated'] = datetime.now().isoformat()
    knowledge_base['total_protocols'] = len(knowledge_base['protocols'])
    
    # Save updated knowledge base
    kb_path.parent.mkdir(parents=True, exist_ok=True)
    with open(kb_path, 'w', encoding='utf-8') as f:
        json.dump(knowledge_base, f, indent=2, ensure_ascii=False)
    
    print(f"\n✅ Knowledge base updated!")
    print(f"   New protocols added: {new_count}")
    print(f"   Total protocols: {knowledge_base['total_protocols']}")
    print(f"   Saved to: {kb_file}")


def main():
    """Main execution"""
    import argparse
    
    parser = argparse.ArgumentParser(description='Update knowledge base with new protocols')
    parser.add_argument('--avengers', default='avengers_cosmic_protocols_complete.txt',
                       help='Avengers protocols file')
    parser.add_argument('--characters', default='character_deep_dives_complete.txt',
                       help='Character protocols file')
    parser.add_argument('--kb', default='processed/knowledge_base.json',
                       help='Knowledge base file')
    
    args = parser.parse_args()
    
    update_knowledge_base(args.avengers, args.characters, args.kb)


if __name__ == '__main__':
    main()
```

### File: `validate_protocols.py`

```python
#!/usr/bin/env python3
"""
Validate that all protocols meet quality standards.

Checks:
- All required fields present
- Quality scores in valid range
- Dimensions adequately detailed
- Business applications specific enough
- Metadata complete
"""

import json
from pathlib import Path
from typing import Dict, List, Tuple


class ProtocolValidator:
    """Validates protocol completeness and quality"""
    
    REQUIRED_FIELDS = [
        'name', 'storyline', 'core_metaphor',
        'd1_bio', 'd2_tech', 'd3_eco', 'd4_cosmic',
        'business_applications', 'warning_signs', 'counter_strategies'
    ]
    
    MIN_QUALITY_SCORE = 0.6
    MIN_DIMENSION_LENGTH = 50  # characters
    MIN_APPLICATIONS = 3
    
    def validate_protocol(self, protocol: Dict) -> Tuple[bool, List[str]]:
        """Validate single protocol, return (is_valid, errors)"""
        errors = []
        
        # Check required fields
        for field in self.REQUIRED_FIELDS:
            if field not in protocol or not protocol[field]:
                errors.append(f"Missing required field: {field}")
        
        # Check quality score
        score = protocol.get('quality_score', 0)
        if score < self.MIN_QUALITY_SCORE:
            errors.append(f"Quality score {score} below minimum {self.MIN_QUALITY_SCORE}")
        
        # Check dimension completeness
        for dim in ['d1_bio', 'd2_tech', 'd3_eco', 'd4_cosmic']:
            content = protocol.get(dim, '')
            if len(content) < self.MIN_DIMENSION_LENGTH:
                errors.append(f"{dim} too short: {len(content)} chars (min {self.MIN_DIMENSION_LENGTH})")
        
        # Check business applications
        apps = protocol.get('business_applications', [])
        if len(apps) < self.MIN_APPLICATIONS:
            errors.append(f"Only {len(apps)} applications (min {self.MIN_APPLICATIONS})")
        
        # Check metadata
        if not protocol.get('risk_severity'):
            errors.append("Missing risk severity")
        
        if not protocol.get('industry_tags'):
            errors.append("Missing industry tags")
        
        return (len(errors) == 0, errors)
    
    def validate_knowledge_base(self, kb_path: str) -> Dict:
        """Validate entire knowledge base"""
        with open(kb_path, 'r', encoding='utf-8') as f:
            kb = json.load(f)
        
        results = {
            'total': len(kb['protocols']),
            'valid': 0,
            'invalid': 0,
            'errors_by_protocol': {}
        }
        
        for protocol in kb['protocols']:
            name = protocol.get('name', 'Unknown')
            is_valid, errors = self.validate_protocol(protocol)
            
            if is_valid:
                results['valid'] += 1
            else:
                results['invalid'] += 1
                results['errors_by_protocol'][name] = errors
        
        return results
    
    def print_report(self, results: Dict):
        """Print validation report"""
        print("\n" + "="*70)
        print("PROTOCOL VALIDATION REPORT")
        print("="*70)
        print(f"\nTotal Protocols: {results['total']}")
        print(f"Valid: {results['valid']} ({results['valid']/results['total']*100:.1f}%)")
        print(f"Invalid: {results['invalid']}")
        
        if results['errors_by_protocol']:
            print(f"\n{'-'*70}")
            print("ERRORS BY PROTOCOL:")
            print(f"{'-'*70}")
            
            for name, errors in results['errors_by_protocol'].items():
                print(f"\n❌ {name}")
                for error in errors:
                    print(f"   - {error}")
        
        print(f"\n{'='*70}\n")


def main():
    """Main execution"""
    import argparse
    
    parser = argparse.ArgumentParser(description='Validate protocol quality')
    parser.add_argument('--kb', default='processed/knowledge_base.json',
                       help='Knowledge base file')
    
    args = parser.parse_args()
    
    validator = ProtocolValidator()
    results = validator.validate_knowledge_base(args.kb)
    validator.print_report(results)
    
    # Exit with error if any protocols invalid
    if results['invalid'] > 0:
        exit(1)


if __name__ == '__main__':
    main()
```
